.386
.model flat, stdcall
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;includem biblioteci, si declaram ce functii vrem sa importam
includelib msvcrt.lib
extern exit: proc
extern printf: proc
extern scanf: proc
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;declaram simbolul start ca public - de acolo incepe executia
public start
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;sectiunile programului, date, respectiv cod
.data

cin_nou dq 5.90
trei dq 3.0
one dq 1.0
zt_do dq 13.28

b dq 0
x dq 0
y dq 0
z dq 0

M1 dq 0
rad dq 0
M2 dq 0
rez dq 0

result dd 0.0      

format DB "%lf", 0
format_af DB "Rez=%lf",13,10, 0
.code
start:
	
	push offset b
	push offset format
	call scanf
	add esp, 8
	
	push dword ptr [b +4]
	push dword ptr [b]
	push offset format_af
	call printf
	add esp, 12
	
	
	
	push offset x
	push offset format
	call scanf
	add esp, 8
	
	
	push dword ptr [x +4]
	push dword ptr [x]
	push offset format_af
	call printf
	add esp, 12
	
	push offset y
	push offset format
	call scanf
	add esp, 8
	
	
	push dword ptr [y +4]
	push dword ptr [y]
	push offset format_af
	call printf
	add esp, 12
	
	push offset z
	push offset format
	call scanf
	add esp, 8
	
	
	push dword ptr [z +4]
	push dword ptr [z]
	push offset format_af
	call printf
	add esp, 12
	
	finit
	
	
	;ctrl + k
	;ctrl + shift + k
	;−b^3.00
	fld trei
	fld b
	fyl2x
	fld st(0)
	fld st(0)
	fistp result ;<- rotunjeste la cea mai apropiata variabila
	fild result  ; si asa
	;frndint <- rotunjeste la cea mai apropiata variabila
	
	; fcom result, st(1)
	; jng salt
	; fld1
	; fsub
	; salt:
	
	fsub
	f2xm1
	fld1
	fadd
	fscale
	
	
	; fld b
	; fld b
	; fmul
	; fld b
	; fmul
	
	
	
	fstp rad
	
	push dword ptr [rad +4]
	push dword ptr [rad]
	push offset format_af
	call printf
	add esp, 12
	
	fld rad
	
	;5.90 + z 
	fld cin_nou
	fld z
	fadd
	fsubr
	
	;5.90 + z −b^3.00
	
	fstp M1
	
	push dword ptr [M1 +4]
	push dword ptr [M1]
	push offset format_af
	call printf
	add esp, 12
	
	
	finit
	
	fld trei
	fld x
	fyl2x
	fld st(0)
	fld st(0)
	frndint
	fsub
	f2xm1
	fld1
	fadd
	fscale
	
	; fld x
	; fld x
	; fmul
	; fld x
	; fmul
	
	
	; 13.28 * y
	fld zt_do
	fld y
	fmul
	
	fdiv
	
	fst M2
	
	
	push dword ptr [M2 +4]
	push dword ptr [M2]
	push offset format_af
	call printf
	add esp, 12
	
	
	fld M1
	fmul
	fstp rez
	
	
	
	push dword ptr [rez +4]
	push dword ptr [rez]
	push offset format_af
	call printf
	add esp, 12
	
	
	
	
	
	push 0
	call exit
end start
