.386
.model flat, stdcall
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;includem msvcrt.lib, si declaram ce functii vrem sa importam
includelib msvcrt.lib
extern exit: proc
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;declaram simbolul start ca public - de acolo incepe executia
public start
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;

;sectiunile programului, date, respectiv cod
.data
mat dw 1,-2,3
    dw 4,5,-6
    dw 2,9,12
    dw 23,11,13
    dw -24,20,36


M dd 5
N dd 3

dim_mat dd ?

k dw 6
sum dw 0 

suma_el dw 5 dup(0)
   
.code
start:
	
	
	mov eax, M
	mul N
	mov ecx, eax ; ecx = n * m repetari
	mov dim_mat, ecx
	
	
	mov eax, M
	mov ebx, 2
	mul ebx
	mov M, eax
	
	mov eax, N
	mov ebx, 2
	mul ebx
	mov N, eax
	
	mov esi, 0
	mov eax, 0
	mov ebx, 0
	mov edi, 0
	mov eax, 0
	mov esp, 0
	

	et_lin:
	mov edx, 0
	mov ax ,mat[esi][edi]
	cwde
	
	cmp eax, 0
	jge nr_pozitiv
	not eax
	add eax, 1
	
	nr_pozitiv:
	div k
	cmp edx, 0
	jnz salt
	
	mov dx, mat[esi][edi]
	add sum, dx
	

	salt:
	inc edi
	inc edi
	
	cmp edi, N
	jnz urm_el;daca elementul curent este pe pozitia finala sau nu din linie
	
	cmp esi, M
	jz final_loop;daca elementul curent este pe pozitia finala sau nu din line si coloana
	
	;inc edi
	;inc edi
	
	add esi, edi
	mov edi, 0 ;trec la primul element de pe urmatoarea linie
	
	jmp lin_noua
	
	urm_el:	
	lin_noua:
	loop et_lin
	
	
	final_loop:
	mov ax, sum
	
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;
;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;;	
	
	; subpunctul b
	mov esi, 0
	mov eax, 0
	mov ebx, 0
	mov edi, 0
	mov eax, 0
	mov esp, 0
	
	mov ecx, dim_mat
	
	et_lin1:
	
	mov eax, 0
	mov ax ,mat[esi][edi]
	
	cwde
	cmp eax, 0
	jnge nr_negativ	
	
	mov dx, mat[esi][edi]
	add suma_el[esp], dx
	mov eax, 0
	mov ax, suma_el[esp]
	
	nr_negativ:
	
	inc edi
	inc edi
	
	cmp edi, N
	jnz urm_el1;daca elementul curent este pe pozitia finala sau nu din linie
	
	cmp esi, M
	jz final_loop1;daca elementul curent este pe pozitia finala sau nu din line si coloana
	
	;inc edi
	;inc edi	
	
	add esi, edi
	mov edi, 0
	
	add esp, 2; pun contorul pe +1 la suma_el la fiecare rand nou
	
	;trec la primul element de pe urmatoarea linie
	jmp lin_noua1
	
	urm_el1:
	lin_noua1:
	loop et_lin1
	
	final_loop1:
	
	mov ax, suma_el[0]
	mov bx, suma_el[2]
	mov cx, suma_el[4]
	mov dx, suma_el[6]
	mov ax, suma_el[8]
	
	;apel functie exit
	push 0
	call exit
end start
