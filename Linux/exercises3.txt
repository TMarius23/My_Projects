#define PIPE_RESPONSE "RESP_PIPE_12990"
#define PIPE_REQUEST "REQ_PIPE_12990"

void constructMessage(char msg[250]){
    int i = 0;
    while(msg[i] != '!' && msg[i] != '\0' && i < 250){
        i++;
    }
    msg[i] = '\0';
}

int main()

        printf("ERROR\ncannot create the response pipe | cannot open the request pipe\n");
        return -1;
    }

    fdReq = open(PIPE_REQUEST, O_RDONLY);
    if(fdReq == -1){
        printf("ERROR\ncannot create the response pipe | cannot open the request pipe\n");
        return -2;
    }

                write(fdResp, "CREATE_SHM!", strlen("CREATE_SHM!"));                
                write(fdResp, "ERROR!", strlen("ERROR!"));
                
                continue;
            }
            
            ftruncate(fd, valNumber);

            
            sharedMem = (volatile char *)mmap(0, valNumber, PROT_READ | PROT_WRITE, MAP_SHARED, fd, 0);
            if(sharedMem == (void *)-1){
                write(fdResp, "CREATE_SHM!", strlen("CREATE_SHM!"));                
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }

            write(fdResp, "CREATE_SHM!", strlen("CREATE_SHM!"));                
            write(fdResp, "SUCCESS!", strlen("SUCCESS!"));

            // printf("Offset%d", offsetV);

                write(fdResp, "WRITE_TO_SHM!", strlen("WRITE_TO_SHM!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));

                    fileName[i] = '\0';
                    break;
                }

                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;

                write(fdResp, "ERROR!", strlen("ERROR!"));

                write(fdResp, "READ_FROM_FILE_OFFSET!", strlen("READ_FROM_FILE_OFFSET!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }

            if(sharedMem == (void *)-1 || mapFile == (void *)-1){
                write(fdResp, "READ_FROM_FILE_OFFSET!", strlen("READ_FROM_FILE_OFFSET!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }

            for(int i = 0; i < noBytesP7; i++){
                sharedMem[i] = mapFile[offsetP7 + i];
            }
  
            write(fdResp, "READ_FROM_FILE_OFFSET!", strlen("READ_FROM_FILE_OFFSET!"));
            write(fdResp, "SUCCESS!", strlen("SUCCESS!"));
        }

        if(strcmp(msg1, "READ_FROM_FILE_SECTION") == 0){
            printf("\n");
            unsigned int sectNo = 0, off = 0, noB = 0;
            read(fdReq, &sectNo, sizeof(sectNo));
            read(fdReq, &off, sizeof(off));
            read(fdReq, &noB, sizeof(noB));
            
           // write(fdResp, "1!", strlen("1!"));s
        

            if(mapFile == (void *)-1){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }
           // write(fdResp, "2!", strlen("2!"));

            if(sharedMem == (void *)-1){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }
           // write(fdResp, "3!", strlen("3!"));

            ///Validare SF
            char magic[3];

            magic[0] = (char)mapFile[sizeFisier - 2];
            magic[1] = (char)mapFile[sizeFisier - 1];
            
           // write(fdResp, "4!", strlen("4!"));


            printf("Magic: %c%c", magic[0], magic[1]);
            

            if(magic[0] != '3' || magic[1] != 'E'){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }
            printf("Pass1");

            unsigned int headerSize = 0;

            headerSize = (u_int8_t)(mapFile[sizeFisier - 3] << 8) |  (u_int8_t)mapFile[sizeFisier - 4];
            printf("Headersz: %d %d %d",headerSize, mapFile[sizeFisier - 3], mapFile[sizeFisier - 4]);

            // if(headerSize < 0){
            //    headerSize = ~headerSize; 
            // }
            printf("SizeFisier:%d Pass2",sizeFisier);

            unsigned int startFileHeader = sizeFisier - headerSize;

            u_int8_t version = mapFile[startFileHeader];
            printf("Version:%d", version);
            if(version < 42 || version > 99){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }

            u_int8_t noOfSect = mapFile[startFileHeader + 1];
            printf("NumberOfSect: %d", noOfSect);
            // if(noOfSect != 48 && noOfSect != 51 && noOfSect != 91 && noOfSect != 60 && noOfSect != 43 && noOfSect != 94){
           if(noOfSect != 2){
                if(noOfSect < 6 || noOfSect > 18){
                    write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                    write(fdResp, "ERROR!", strlen("ERROR!"));
                    continue;
                }
            }
            printf("Number of sect fisier: %d, ceruti: %d", noOfSect, sectNo);

            if(noOfSect < sectNo){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }


            unsigned int sectNoStart = startFileHeader + 2;

            // int verificareType = 1;
            // for(int i = 1; i <= noOfSect && verificareType == 1; i++){
            //     sectNoStart = sectNoStart + 11;
                
            //     short typeSect = mapFile[sectNoStart + 1] << 8 | mapFile[sectNoStart];
            //     if(typeSect != 48 && typeSect != 51 && typeSect != 91 && typeSect != 60 && typeSect != 43 && typeSect != 94){
            //         verificareType = 0;
            //     }
            //     sectNoStart = sectNoStart + 10;
            // }

            // printf("Verificare:%d", verificareType);

            // if(verificareType == 0){
            //     write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
            //     write(fdResp, "ERROR!", strlen("ERROR!"));
            //     continue;
            // }   


            sectNoStart = startFileHeader + 2;
            printf("StartCountSection:%d",sectNoStart);

            for(int i = 1; i < sectNo; i++){//stiu caa e fara dar momentan verific ceva
                sectNoStart = sectNoStart + 21;
            }

            sectNoStart = sectNoStart + 13;
            printf("SECTIONSTART:%d",sectNoStart);
            
            unsigned int citireFileUO = (u_int8_t)mapFile[sectNoStart + 3] << 24 |  (u_int8_t)mapFile[sectNoStart + 2] << 16 |
                                                                    (u_int8_t)mapFile[sectNoStart + 1] << 8 |  (u_int8_t)mapFile[sectNoStart];

            sectNoStart = sectNoStart + 4;

                                                                    (u_int8_t)mapFile[sectNoStart + 1] << 8 |  (u_int8_t)mapFile[sectNoStart];
            printf("SizeSectiuneStart:%d ", sizeSectR);


            if(sizeSectR < (off + noB)){
                write(fdResp, "READ_FROM_FILE_SECTION!", strlen("READ_FROM_FILE_SECTION!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;
            }

            ///citirea se face invers, sirul e pus pe dos in SF
            citireFileUO = citireFileUO + off;

        }

        if(strcmp(msg1, "READ_FROM_LOGICAL_SPACE_OFFSET") == 0){
            unsigned int logicalOffset = 0, noOfBytes = 0;
            read(fdReq, &logicalOffset, sizeof(logicalOffset));
            read(fdReq, &noOfBytes, sizeof(noOfBytes));

            printf("Magic: %c%c", magic[0], magic[1]);
            

            if(magic[0] != '3' || magic[1] != 'E'){
                write(fdResp, "READ_FROM_LOGICAL_SPACE_OFFSET!", strlen("READ_FROM_LOGICAL_SPACE_OFFSET!"));

                    write(fdResp, "ERROR!", strlen("ERROR!"));

            unsigned int sectNoStart = startFileHeader + 2;
            unsigned offsetLogicalModulo = logicalOffset % aliniament4096;
            logicalOffset = logicalOffset / aliniament4096;
            logicalOffset *= aliniament4096;
            printf("LogicalOffset:%d", logicalOffset);

                                            (u_int8_t)mapFile[sectNoStart + 1] << 8  | (u_int8_t)mapFile[sectNoStart];
                printf("SizeSectiune: %d", sizeSectiune);
                
                sizeSectiune = sizeSectiune / aliniament4096;
                contorMemorie += aliniament4096 * sizeSectiune;
                contorMemorie += aliniament4096;
                sectNoStart += 4;
            }

            if(ok == 0){
                write(fdResp, "READ_FROM_LOGICAL_SPACE_OFFSET!", strlen("READ_FROM_LOGICAL_SPACE_OFFSET!"));
                write(fdResp, "ERROR!", strlen("ERROR!"));
                continue;

            sectNoStart += 13;

            }

            write(fdResp, "READ_FROM_LOGICAL_SPACE_OFFSET!", strlen("READ_FROM_LOGICAL_SPACE_OFFSET!"));
            write(fdResp, "SUCCESS!", strlen("SUCCESS!"));
        }

        if(strcmp(msg1, "EXIT") == 0){
           break;
        }
   }

    close(fdReq);
    close(fdResp);
    unlink(PIPE_RESPONSE);
